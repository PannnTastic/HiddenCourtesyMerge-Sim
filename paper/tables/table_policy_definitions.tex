% Table 6: Policy definitions
% Source: evaluate_policies.py docstring and policy_summary.json notes
\begin{table}[htbp]
\centering
\caption{Policy baselines evaluated in this work.
  All policies share the same environment seed, hidden courtesy type,
  traffic density, and observation noise per episode index.
  Action sampling uses separate per-policy RNG offsets to avoid
  cross-contamination of the exogenous scenario stream.}
\label{tab:policy_defs}
\small
\begin{tabular}{lp{8.5cm}}
\toprule
\textbf{Policy} & \textbf{Description} \\
\midrule
\texttt{random\_policy}
  & Selects an action uniformly at random from all five meta-actions at every
    step. Provides an unconditional lower bound. \\
\addlinespace
\texttt{rule\_policy}
  & Uses the same heuristic action table as \texttt{belief\_policy} but with
    the belief fixed at the uniform prior $b=[0.5,\,0.5]$ throughout the
    episode. Isolates the marginal value of belief estimation. \\
\addlinespace
\texttt{belief\_policy}
  & Maintains a Bayesian belief over the hidden courtesy type and passes it to
    the heuristic action table. Belief is updated only while the merge vehicle
    is inside the interaction window (2--40\,m ahead). \\
\addlinespace
\texttt{oracle\_policy}
  & Passes a one-hot belief at the true hidden courtesy type to the same
    heuristic action table. Constitutes an \emph{upper bound for this action
    table only}, not for the full POMDP policy space; optimal POMDP solving
    is not claimed. \\
\addlinespace
\texttt{pomcp\_policy}
  & UCT-based POMCP solver (100 simulations/step, horizon 10 steps, $\gamma{=}0.98$).
    Samples the hidden mode from the external Bayesian belief each simulation.
    Generative model uses mean IDM parameters per courtesy type.
    Action space restricted to \{IDLE, SLOWER\} for comparability with
    heuristic policies. \\
\bottomrule
\end{tabular}
\end{table}
